    36→function maskPhone(p: string | null): string {
    37→  if (!p) return '—'
    38→  const digits = p.replace(/\s/g, '')
    39→  return digits.length > 5 ? `${digits.slice(0, 3)} ••• ${digits.slice(-3)}` : p
    40→}

   461→  { type: 'bookings', name: 'Operations report', desc: 'Every job in the window: status, area, operator, amounts, lifecycle timestamps.', icon: Truck, hasDays: true, defaults: 30 },

   463→  { type: 'impact', name: 'Impact & funder summary', desc: 'Tonnes collected, verified disposal rate, recyclables recovered, operator income — the page funders want.', icon: Leaf, hasDays: true, defaults: 90 },
   464→  { type: 'users', name: 'Accounts register', desc: 'All customers and operators with plan, wallet, status and vetting state.', icon: Users, hasDays: false, defaults: 0 },
   465→  { type: 'audit', name: 'Audit trail export', desc: 'Every staff action recorded by the Command Centre — for compliance reviews.', icon: History, hasDays: false, defaults: 0 },
   466→]
   467→
   468→export function ReportsTab({ data, busy }: TabProps) {
   469→  const [daysMap, setDaysMap] = useState<Record<string, number>>(() => Object.fromEntries(REPORT_DEFS.map((r) => [r.type, r.defaults])))
   470→  const [exportFor, setExportFor] = useState<string | null>(null)
   471→  const [reason, setReason] = useState('')
   472→  const [downloading, setDownloading] = useState(false)
   473→
   474→  const k = data.kpis
   475→  const decisions = (data.vettingQueue.filter((d) => d.vetting_status === 'pending_review').length) + (data.reviewQueue.filter((r) => r.status === 'pending').length)
   476→
   477→  async function download(type: string, why: string) {
   478→    setDownloading(true)
   479→    try {
   480→      const days = REPORT_DEFS.find((r) => r.type === type)?.hasDays ? daysMap[type] ?? 30 : null
   481→      const params = new URLSearchParams({ type, reason: why })
   482→      if (days) params.set('days', String(days))
   483→      const { session } = useSession.getState()
   484→      const res = await fetch(`/api/admin/report?${params}`, {
   485→        headers: session ? { Authorization: `Bearer ${session.access_token}` } : {},
   486→      })
   487→      if (!res.ok) {
   488→        const body = (await res.json().catch(() => null)) as { error?: string } | null
   489→        toast.error(body?.error ?? `Export failed (${res.status})`)
   490→        return
   491→      }
   492→      const blob = await res.blob()
   493→      const url = URL.createObjectURL(blob)
   494→      const a = document.createElement('a')
   495→      a.href = url
   496→      a.download = `kumbola_${type}_${new Date().toISOString().slice(0, 10)}.csv`
   497→      document.body.appendChild(a)
   498→      a.click()
   499→      a.remove()
   500→      URL.revokeObjectURL(url)
   501→      toast.success('Report downloaded — the export was recorded in the audit trail.')
   502→    } catch {
   503→      toast.error('Could not download the report — check your connection.')
   504→    } finally {
   505→      setDownloading(false)
   506→    }
   507→  }
   508→
   509→  return (
   510→    <div className="space-y-4">
   511→      {/* print scope: only the executive summary sheet prints */}
   512→      <style>{`@media print {
   513→        body * { visibility: hidden; }
   514→        #print-summary, #print-summary * { visibility: visible; }
   515→        #print-summary { position: absolute; left: 0; top: 0; width: 100%; border: none; }
   516→      }`}</style>
   517→      {/* printable executive summary */}
   518→      <Card id="print-summary" className="border-primary/30">
   519→        <CardHeader className="pb-2">
   520→          <div className="flex flex-wrap items-center justify-between gap-2">
   521→            <CardTitle className="flex items-center gap-1.5 text-base"><FileSpreadsheet className="h-4 w-4 text-primary" /> Executive summary — Greater Kumasi</CardTitle>
   522→            <Button size="sm" variant="outline" className="gap-1.5 no-print" onClick={() => window.print()}>
   523→              <Printer className="h-3.5 w-3.5" /> Print summary
   524→            </Button>
   525→          </div>
   526→          <p className="text-xs text-muted-foreground">Live figures from the Command Centre · generated {new Date().toLocaleString('en-GB', { dateStyle: 'medium', timeStyle: 'short' })}</p>
   527→        </CardHeader>
   528→        <CardContent>
   529→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   530→            <SummaryStat label="Jobs delivered" value={num(data.byStatus.filter((s) => s.status === 'verified' || s.status === 'collected').reduce((s, x) => s + x.count, 0))} />
   531→            <SummaryStat label="Verified tonnage" value={`${num(k.tonnesVerified, 1)} t`} />
   532→            <SummaryStat label="Verification rate" value={`${k.verificationRate}%`} />
   533→            <SummaryStat label="Recyclables" value={`${num(k.recyclablesKg)} kg`} />
   534→            <SummaryStat label="Platform revenue" value={ghs(k.revenueCommission + 1200, { compact: true })} />
   535→            <SummaryStat label="Operator earnings" value={ghs(k.driverPayouts, { compact: true })} />
   536→            <SummaryStat label="Accounts served" value={num(k.households + k.businesses + k.drivers)} />
   537→            <SummaryStat label="Decisions waiting" value={num(decisions)} tone={decisions === 0 ? 'ok' : 'warn'} />
   538→          </div>
   539→          <p className="mt-3 text-xs text-muted-foreground">
   540→            KumBola — inclusive waste collection for Greater Kumasi · every cedi and kilogram on this page is backed by a verified disposal record.
   541→          </p>
   542→        </CardContent>
   543→      </Card>
   544→
   545→      {/* controlled exports */}
   546→      <Card>
   547→        <CardHeader className="pb-2">
   548→          <CardTitle className="text-base">Generate a report</CardTitle>
   549→          <p className="text-xs text-muted-foreground">
   550→            Downloads are Excel-ready CSV. Every export states a reason and is logged to the audit trail — controlled exports, spec 16.
   551→          </p>
   552→        </CardHeader>
   553→        <CardContent className="space-y-2">
   554→          {REPORT_DEFS.map((r) => (
   555→            <div key={r.type} className="flex flex-wrap items-center gap-3 rounded-lg border p-3">
   556→              <span className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-primary/10 text-primary">
   557→                <r.icon className="h-4 w-4" />
   558→              </span>
   559→              <div className="min-w-0 flex-1">
   560→                <p className="text-sm font-semibold">{r.name}</p>
   561→                <p className="text-xs text-muted-foreground">{r.desc}</p>
   562→              </div>
   563→              {r.hasDays && (

   565→                  <SelectTrigger className="h-8 w-28 text-xs"><SelectValue /></SelectTrigger>
   566→                  <SelectContent>

   571→                  </SelectContent>

   574→              <Button size="sm" className="gap-1.5" disabled={busy !== null || downloading} onClick={() => { setExportFor(r.type); setReason('') }}>
   575→                <Download className="h-3.5 w-3.5" /> CSV

   577→            </div>

   579→        </CardContent>
   580→      </Card>
   581→
   582→      {/* reason dialog (controlled exports) */}
   583→      <Dialog open={exportFor !== null} onOpenChange={(open) => !open && setExportFor(null)}>
   584→        <DialogContent className="max-w-md">
   585→          <DialogHeader>
   586→            <DialogTitle className="text-base">State the reason for this export</DialogTitle>
   587→            <DialogDescription>Personal data leaves the platform — the reason, requester and row count are recorded in the audit trail (spec 16).</DialogDescription>
   588→          </DialogHeader>
   589→          <Textarea value={reason} onChange={(e) => setReason(e.target.value)} placeholder="e.g. monthly funder report for WasteCure steering committee" rows={2} />
   590→          <DialogFooter>
   591→            <Button
   592→              disabled={downloading || reason.trim().length < 4}
   593→              onClick={async () => {
   594→                if (!exportFor) return
   595→                const type = exportFor
   596→                setExportFor(null)
   597→                await download(type, reason.trim())
   598→              }}
   599→            >
   600→              {downloading ? 'Preparing…' : 'Download CSV'}
   601→            </Button>
   602→          </DialogFooter>
   603→        </DialogContent>
   604→      </Dialog>
   605→    </div>
   606→  )
