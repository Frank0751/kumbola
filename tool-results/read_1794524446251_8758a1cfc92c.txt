   796→                      <p className="text-sm font-medium">{d.full_name}</p>
   797→                      <p className="text-xs text-muted-foreground">{d.vehicle_plate ?? '—'} · {d.base_municipality ?? d.municipality} · {d.rating.toFixed(1)} ★ · {d.total_jobs} jobs</p>
   798→                    </div>
   799→                    {busy === `dispatch-${dispatchFor.id}` && <Loader2 className="h-4 w-4 animate-spin" />}
   800→                  </button>
   801→                ))}
   802→              </div>
   803→            </>
   804→          )}
   805→        </DialogContent>
   806→      </Dialog>
   807→
   808→      {/* ---------- municipality drill-down ---------- */}
   809→      <Dialog open={drill !== null} onOpenChange={(open) => !open && setDrill(null)}>
   810→        <DialogContent className="max-w-md">
   811→          <DialogHeader>
   812→            <DialogTitle className="text-base">{drill} — recent jobs</DialogTitle>
   813→            <DialogDescription>From the most recent 500 bookings.</DialogDescription>
   814→          </DialogHeader>
   815→          <div className="max-h-80 space-y-1.5 overflow-y-auto text-sm">
   816→            {data?.recentBookings.filter((b) => b.municipality === drill).map((b) => (
   817→              <div key={b.id} className="flex items-center justify-between gap-2 rounded-lg border p-2 text-xs">
   818→                <span className="font-mono">{b.reference}</span>
   819→                <span className="min-w-0 flex-1 truncate text-muted-foreground">{b.customer?.full_name ?? '—'} · {b.driver?.full_name ?? 'unassigned'}</span>
   820→                <StatusBadge status={b.status} />
   821→              </div>
   822→            ))}
   823→            {data && data.recentBookings.filter((b) => b.municipality === drill).length === 0 && (
   824→              <p className="py-4 text-center text-muted-foreground">No recent jobs here.</p>
   825→            )}
   826→          </div>
   827→        </DialogContent>
   828→      </Dialog>
   829→
   830→      {/* ---------- reject / suspend note dialog ---------- */}
   831→      <Dialog open={noteFor !== null} onOpenChange={(open) => !open && setNoteFor(null)}>
   832→        <DialogContent className="max-w-md">
   833→          {noteFor && (
   834→            <>
   835→              <DialogHeader>
   836→                <DialogTitle className="text-base">
   837→                  {noteFor.kind === 'vet'
   838→                    ? `${noteFor.decision === 'reject' ? 'Reject' : 'Suspend'} ${noteFor.driver.full_name}`
   839→                    : `Reject disposal proof — ${noteFor.reference}`}
   840→                </DialogTitle>
   841→                <DialogDescription>The note is delivered to the operator and recorded in the audit trail.</DialogDescription>
   842→              </DialogHeader>
   843→              <Textarea value={noteText} onChange={(e) => setNoteText(e.target.value)} placeholder="Reason — e.g. licence expired, documents unreadable, photo does not show the site…" rows={3} />
   844→              <DialogFooter>
   845→                <Button
   846→                  disabled={busy !== null || noteText.trim().length < 4}
   847→                  onClick={async () => {
   848→                    if (noteFor.kind === 'vet') {
   849→                      await act({ action: 'vet_driver', driverId: noteFor.driver.id, decision: noteFor.decision, note: noteText.trim() }, 'note')
   850→                    } else {
   851→                      await act({ action: 'review_disposal', verificationId: noteFor.id, decision: 'reject', note: noteText.trim() }, 'note')
   852→                    }
   853→                    setNoteFor(null)
   854→                  }}
   855→                >
   856→                  {busy === 'note' ? <Loader2 className="h-4 w-4 animate-spin" /> : 'Confirm'}
   857→                </Button>
   858→              </DialogFooter>
   859→            </>
   860→          )}
   861→        </DialogContent>
   862→      </Dialog>
   863→    </div>
   864→  )
   865→}
