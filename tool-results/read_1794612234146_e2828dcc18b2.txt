   849→                    ))}

   851→                </table>
   852→              </CardContent>
   853→            </Card>
   854→          </TabsContent>
   855→        </Tabs>
   856→      </main>
   857→
   858→      {/* ---------- job detail dialog ---------- */}

   921→                    key={d.id}

   927→                    }}

   929→                    <Avatar className="h-8 w-8">
   930→                      <AvatarFallback className="bg-primary/10 text-[10px] font-bold text-primary">{initials(d.full_name)}</AvatarFallback>

   935→                    </div>
   936→                    {busy === `dispatch-${dispatchFor.id}` && <Loader2 className="h-4 w-4 animate-spin" />}
   937→                  </button>
   938→                ))}
   939→              </div>
   940→            </>
   941→          )}
   942→        </DialogContent>
   943→      </Dialog>
   944→
   945→      {/* ---------- municipality drill-down ---------- */}

   960→            {data && data.recentBookings.filter((b) => b.municipality === drill).length === 0 && (
   961→              <p className="py-4 text-center text-muted-foreground">No recent jobs here.</p>
   962→            )}
   963→          </div>
   964→        </DialogContent>
   965→      </Dialog>
   966→
   967→      {/* ---------- reject / suspend note dialog ---------- */}
   968→      <Dialog open={noteFor !== null} onOpenChange={(open) => !open && setNoteFor(null)}>
   969→        <DialogContent className="max-w-md">
   970→          {noteFor && (
   971→            <>
   972→              <DialogHeader>
   973→                <DialogTitle className="text-base">
   974→                  {noteFor.kind === 'vet'
   975→                    ? `${noteFor.decision === 'reject' ? 'Reject' : 'Suspend'} ${noteFor.driver.full_name}`
   976→                    : `Reject disposal proof — ${noteFor.reference}`}
   977→                </DialogTitle>
   978→                <DialogDescription>The note is delivered to the operator and recorded in the audit trail.</DialogDescription>
   979→              </DialogHeader>
   980→              <Textarea value={noteText} onChange={(e) => setNoteText(e.target.value)} placeholder="Reason — e.g. licence expired, documents unreadable, photo does not show the site…" rows={3} />
   981→              <DialogFooter>
   982→                <Button
   983→                  disabled={busy !== null || noteText.trim().length < 4}
   984→                  onClick={async () => {
   985→                    if (noteFor.kind === 'vet') {
   986→                      const ok = await act({ action: 'vet_driver', driverId: noteFor.driver.id, decision: noteFor.decision, note: noteText.trim() }, 'note')
   987→                      if (ok) setNoteFor(null)
   988→                    } else {
   989→                      const ok = await act({ action: 'review_disposal', verificationId: noteFor.id, decision: 'reject', note: noteText.trim() }, 'note')
   990→                      if (ok) setNoteFor(null)
   991→                    }
   992→                  }}
   993→                >
   994→                  {busy === 'note' ? <Loader2 className="h-4 w-4 animate-spin" /> : 'Confirm'}
   995→                </Button>
   996→              </DialogFooter>
   997→            </>
   998→          )}
   999→        </DialogContent>
  1000→      </Dialog>
  1001→    </div>
  1002→  )
